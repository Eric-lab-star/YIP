    "지역": ["중구", "중구", "강남구", "강남구", "마포구", "마포구"],
    "미세먼지": [30, 34, 42, 38, 25, 29],
}
df = pd.DataFrame(data)

지역평균 = df.groupby("지역")["미세먼지"].mean().reset_index()
sns.____(data=지역평균, x="지역", y="미세먼지")
plt.show()
