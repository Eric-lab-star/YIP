```python
import pandas as pd

data = {
    "연령대": ["10대", "20대", "30대"],
    "남자": [4200, 5100, 4800],
    "여자": [4000, 5300, 4900],
}
df = pd.DataFrame(data)

길게 = pd.melt(
    df,
    id_vars=["연령대"],
    value_vars=____,
    var_name="성별",
    value_name="인구수",
)
print(길게)
```
